// ===== CANVAS BACKGROUND =====
const canvas = document.getElementById('bg');
const ctx = canvas.getContext('2d');
let W, H, stars=[], petals=[], bursts=[];
function resize(){ W=canvas.width=innerWidth; H=canvas.height=innerHeight; }
addEventListener('resize', resize); resize();

for (let i=0;i<120;i++) stars.push({x:Math.random()*W, y:Math.random()*H, r:Math.random()*1.5+.3, tw:Math.random()*Math.PI*2, sp:Math.random()*.02+.005});
for (let i=0;i<25;i++) petals.push({x:Math.random()*W, y:Math.random()*H, r:Math.random()*6+4, vx:Math.random()*.5-.25, vy:Math.random()*.8+.4, rot:Math.random()*Math.PI*2, vr:(Math.random()-.5)*.04, hue:340+Math.random()*20});

function burst(e){
  const x = e?.clientX ?? W/2, y = e?.clientY ?? H/2;
  for (let i=0;i<12;i++){
    const a = Math.random()*Math.PI*2, s = Math.random()*4+2;
    bursts.push({x,y, vx:Math.cos(a)*s, vy:Math.sin(a)*s, life:1, size:Math.random()*12+10});
  }
  const c = document.getElementById('count');
  c.textContent = +c.textContent + 1;
}

function draw(){
  ctx.fillStyle = 'rgba(26,11,31,.25)';
  ctx.fillRect(0,0,W,H);
  // stars
  stars.forEach(s => {
    s.tw += s.sp;
    const a = .4 + Math.sin(s.tw)*.6;
    ctx.fillStyle = `rgba(255,220,240,${a})`;
    ctx.beginPath(); ctx.arc(s.x, s.y, s.r, 0, Math.PI*2); ctx.fill();
  });
  // petals
  petals.forEach(p => {
    p.x += p.vx; p.y += p.vy; p.rot += p.vr;
    if (p.y > H+20) { p.y = -20; p.x = Math.random()*W; }
    if (p.x < -20) p.x = W+20; if (p.x > W+20) p.x = -20;
    ctx.save(); ctx.translate(p.x, p.y); ctx.rotate(p.rot);
    ctx.fillStyle = `hsla(${p.hue},80%,75%,.5)`;
    ctx.beginPath(); ctx.ellipse(0,0,p.r,p.r*.6,0,0,Math.PI*2); ctx.fill();
    ctx.restore();
  });
  // bursts
  bursts = bursts.filter(b => {
    b.x += b.vx; b.y += b.vy; b.vy += .15; b.life -= .02;
    if (b.life <= 0) return false;
    ctx.globalAlpha = b.life;
    ctx.font = b.size+'px serif';
    ctx.fillText('💗', b.x, b.y);
    ctx.globalAlpha = 1;
    return true;
  });
  requestAnimationFrame(draw);
}
draw();

let li=0, ci=0, deleting=false;
function type(){
  const line = typewriterLines[li];
  if (!deleting) {
    tw.textContent = line.slice(0, ++ci);
    if (ci === line.length) { deleting = true; setTimeout(type, 2500); return; }
  } else {
    tw.textContent = line.slice(0, --ci);
    if (ci === 0) { deleting = false; li = (li+1)%typewriterLines.length; }
  }
  setTimeout(type, deleting ? 30 : 70);
}

// ===== MUSIC (Web Audio) =====
let audioCtx, musicOn=false;
document.getElementById('musicBtn').addEventListener('click', () => {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    const notes = [261.63, 220.00, 174.61, 196.00]; // Cmaj7, Am7, Fmaj7, G
    let step = 0;
    function playNote(){
      if (!musicOn) return;
      const freq = notes[step % notes.length];
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = 'sine';
      osc.frequency.value = freq;
      gain.gain.setValueAtTime(0, audioCtx.currentTime);
      gain.gain.linearRampToValueAtTime(.08, audioCtx.currentTime+.3);
      gain.gain.exponentialRampToValueAtTime(.001, audioCtx.currentTime+2.5);
      osc.connect(gain).connect(audioCtx.destination);
      osc.start(); osc.stop(audioCtx.currentTime+2.5);
      // harmony
      const osc2 = audioCtx.createOscillator();
      const gain2 = audioCtx.createGain();
      osc2.type = 'triangle';
      osc2.frequency.value = freq * 1.5;
      gain2.gain.setValueAtTime(0, audioCtx.currentTime);
      gain2.gain.linearRampToValueAtTime(.04, audioCtx.currentTime+.4);
      gain2.gain.exponentialRampToValueAtTime(.001, audioCtx.currentTime+2.5);
      osc2.connect(gain2).connect(audioCtx.destination);
      osc2.start(); osc2.stop(audioCtx.currentTime+2.5);
      step++;
      setTimeout(playNote, 2200);
    }
    playNote();
  }
  musicOn = !musicOn;
  document.getElementById('musicBtn').textContent = musicOn ? '🔊' : '🎵';
});
